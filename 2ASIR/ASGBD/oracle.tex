\subsection{Instancia vs Base de Datos}
\label{sec:org124ca3c}
\begin{itemize}
\item \textbf{Instancia:} Procesos en memoria + Estructuras de memoria (SGA).
\item \textbf{Base de Datos:} Archivos físicos en disco (Datafiles, Control files).
\end{itemize}

\label{sec:org452cf3e}

\label{sec:orgdbc9309}
\begin{itemize}
\item Shared Pool (Library Cache, Dictionary Cache)
\item Database Buffer Cache
\item Redo Log Buffer
\end{itemize}
\subsubsection{PGA (Program Global Area)}
\label{sec:orgfba5e62}
\begin{itemize}
\item Memoria privada para cada proceso servidor.
\end{itemize}
\subsection{Procesos Background}
\label{sec:orgade347c}
\begin{itemize}
\item DBWn (Database Writer): Escribe buffers sucios a disco.
\item LGWR (Log Writer): Escribe el redo log buffer al disco.
\item CKPT (Checkpoint): Actualiza cabeceras de datafiles.
\item PMON y SMON.
\end{itemize}
